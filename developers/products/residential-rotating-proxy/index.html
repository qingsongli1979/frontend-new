package main

import (
    "fmt"
    "io"
    "net/http"
    "net/url"
    "os"
    "time"
)

func main() {
    proxyURL := &url.URL{
        Scheme: "http",
        Host: "residential.123proxy.cn:33000",
        User: url.UserPassword(os.Getenv("PROXY_USER"), os.Getenv("PROXY_PASS")),
    }
    client := &http.Client{
        Transport: &http.Transport{Proxy: http.ProxyURL(proxyURL)},
        Timeout: 30 * time.Second,
    }
    response, err := client.Get("https://api.ipify.org?format=json")
    if err != nil { panic(err) }
    defer response.Body.Close()
    if response.StatusCode >= 400 { panic(response.Status) }
    body, err := io.ReadAll(response.Body)
    if err != nil { panic(err) }
    fmt.Println(string(body))
}
import java.net.Authenticator;
import java.net.InetSocketAddress;
import java.net.PasswordAuthentication;
import java.net.ProxySelector;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class ProxyCheck {
    public static void main(String[] args) throws Exception {
        System.setProperty("jdk.http.auth.tunneling.disabledSchemes", "");
        String host = "residential.123proxy.cn";
        int port = 33000;
        String user = System.getenv("PROXY_USER");
        char[] password = System.getenv("PROXY_PASS").toCharArray();

        HttpClient client = HttpClient.newBuilder()
            .proxy(ProxySelector.of(new InetSocketAddress(host, port)))
            .authenticator(new Authenticator() {
                protected PasswordAuthentication getPasswordAuthentication() {
                    return new PasswordAuthentication(user, password);
                }
            })
            .connectTimeout(Duration.ofSeconds(15))
            .build();

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create("https://api.ipify.org?format=json"))
            .timeout(Duration.ofSeconds(30))
            .GET()
            .build();
        HttpResponse<String> response = client.send(
            request,
            HttpResponse.BodyHandlers.ofString()
        );
        if (response.statusCode() >= 400) throw new RuntimeException(response.body());
        System.out.println(response.body());
    }
}

固定网关不随国家和SESSION变化

主机始终为 residential.123proxy.cn，端口始终为 33000。国家与 SESSION 由完整认证用户名控制。

02
GEO ROUTING

国家在提取代理时选择

购买流量套餐时不锁定地区。每次生成代理时决定全球随机、欧洲随机或具体国家；具体国家使用小写国家码，并始终放在认证用户名最后。

全球随机

认证用户名不添加国家后缀，从全球住宅池随机选择出口。

地区或国家

例如美国添加 +us，日本添加 +jp，国家码必须为小写。

变更生效

首次生成或修改国家、SESSION 后，通常需要约 3–15 分钟同步。

3–15 分钟不是 SESSION 时长

前者是路由配置同步时间；SESSION 时长是你设置的 1–120 分钟粘性窗口，两者互不替代。

04
STICKY SESSION

什么时候使用 SESSION

默认模式下，新请求会进入轮转逻辑。分页、短流程或多步骤请求需要尽量保持同一住宅出口时，再启用 Sticky SESSION。

12 位 SESSION ID

只使用英文字母和数字。复用同一个 ID 才会复用同一条粘性路由。

1–120 分钟

时长到期或出口不可用后，系统会重新分配住宅 IP。

主动换出口

修改 SESSION ID，或切回每次请求轮转模式。

05
AUTHENTICATION

账密认证与套餐 IP 白名单

账密认证

程序发送完整路由用户名与代理密码，国家码和 SESSION 都能直接从用户名看出。

HOST:PORT:ROUTED_USER:PASSWORD
套餐 IP 白名单

白名单绑定当前套餐。程序只使用 HOST:PORT，国家和 SESSION 由控制台保存为套餐路由。

HOST:PORT

白名单输入时每行一个公网 IPv4；保存到后端时会转为英文逗号分隔。更换采集服务器后需更新当前套餐白名单。

08
PRODUCTION

生产使用建议

等待路由同步变更国家或 SESSION 后预留 3–15 分钟，再重新建立连接验证。
按任务生成 SESSION ID不要让无关任务长期共享同一个 SESSION。
统计实际流量响应体、图片和其他资源都会进入流量消耗。
验证国家与有效内容同时检查出口位置、状态码和响应内容，不只判断 HTTP 200。
09
TROUBLESHOOTING

常见问题

国家码应该放在哪里？

国家码必须为小写并放在完整认证用户名最后。例如美国为 proxy-user+us，带 SESSION 时为 proxy-user-sess_12位ID_15+us。

设置国家后为什么没有立即变化？

首次生成或修改国家、SESSION 后通常需要约 3–15 分钟同步。等待后重新建立连接，不要复用旧连接池验证。

白名单模式还需要发送国家码吗？

不需要在程序中发送用户名。国家与 SESSION 会由控制台保存到当前套餐路由，程序只连接生成的 HOST:PORT。

它是否支持并发线程不限流量？

不支持。隧道住宅代理仅按实际传输流量使用；需要并发线程套餐应选择隧道代理。
